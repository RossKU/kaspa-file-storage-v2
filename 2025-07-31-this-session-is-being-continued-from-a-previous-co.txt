  問題の発生フロー

  1. アップロード完了時（13037行）
    - updateIndex()でインデックス追加
  2. メタTx情報更新時（13554行）
    - 同じエントリーでupdateIndex()再実行
    - 古いインデックスが残ったまま新規追加

  - sortIndexes.nameに同一エントリーが複数存在
  - updateFileListFromKENV()が各インデックスに対して行を作成
  - 画面に同じファイルが3回表示される

  修正方針

  updateIndex()実行前に既存インデックスを削除する処理が必要

  // 堅牢なLocalStorage操作
  class SafeStorage {
      static setItem(key, value) {
          try {
              localStorage.setItem(key, value);
              return { success: true };
          } catch (e) {
              if (e.name === 'QuotaExceededError') {
                  // 容量超過時の処理
                  this.handleQuotaExceeded(key);

                  // リトライ
                  try {
                      localStorage.setItem(key, value);
                      return { success: true };
                  } catch (e2) {
                      return { success: false, error: 'quota_exceeded' };
                  }
              }
              return { success: false, error: e.message };
          }
      }

      static getItem(key) {
          try {
              const value = localStorage.getItem(key);
              if (!value) return null;

              // JSON形式の場合は自動パース
              if (value.startsWith('{') || value.startsWith('[')) {
                  try {
                      return JSON.parse(value);
                  } catch (e) {
                      console.error(`Invalid JSON in ${key}:`, e);
                      // 破損データを削除
                      localStorage.removeItem(key);
                      return null;
                  }
              }
              return value;
          } catch (e) {
              console.error(`Storage access error for ${key}:`, e);
              return null;
          }
      }

      static handleQuotaExceeded(newKey) {
          // 優先度の低いデータから削除
          const priorityOrder = [
              'kaspa-upload-progress',     // 一時的
              'kaspa-upload-metadata',     // 一時的
              'kaspaDirectories',          // 再構築可能
              'kaspaDAG',                  // 再構築可能
              'kenv_temp_data'            // 重要（最後に削除）
          ];

          for (const key of priorityOrder) {
              if (key !== newKey) {
                  localStorage.removeItem(key);
                  log(`[Storage] 容量確保のため${key}を削除`, 'warning');
                  break;
              }
          }
      }
  }

  // KENVManagerでの使用例
  async saveToLocalStorage() {
      const data = {
          entries: this.entries,
          kindex: this.kindex,
          timestamp: new Date().toISOString()
      };

      const result = SafeStorage.setItem('kenv_temp_data', JSON.stringify(data));

      if (!result.success) {
          if (result.error === 'quota_exceeded') {
              // 古いエントリーを削除して再試行
              this.reduceEntriesForStorage();
              const retry = SafeStorage.setItem('kenv_temp_data', JSON.stringify({
                  entries: this.entries,
                  kindex: this.kindex,
                  truncated: true
              }));

              if (!retry.success) {
                  log('[KENV] LocalStorage保存失敗 - ワークスペース設定を推奨', 'error');
                  this.showStorageFullWarning();
              }
          }
      }
  }

  // データ整合性チェック
  validateKENVData(data) {
      if (!data || typeof data !== 'object') return false;
      if (!Array.isArray(data.entries)) return false;
      if (!data.kindex || typeof data.kindex !== 'object') return false;

      // インデックスとエントリーの整合性
      const maxLineNumber = data.entries.length - 1;
      for (const [name, lineNum] of data.kindex.sortIndexes.name || []) {
          if (lineNum > maxLineNumber) {
              log('[KENV] インデックス不整合を検出', 'error');
              return false;
          }
      }

      return true;
  }

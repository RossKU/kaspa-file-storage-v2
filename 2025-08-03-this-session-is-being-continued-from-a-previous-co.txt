      if (kenvEntry.status === 'pending' && kenvEntry.metadata) {
          // 生成のみの場合：保存されたmetadataから再構成
          const kaspaData = kenvEntry.metadata;

          // .kaspaファイルとして出力
          const blob = new Blob([JSON.stringify(kaspaData, null, 2)],
                               { type: 'application/json' });
          const url = URL.createObjectURL(blob);
          const a = document.createElement('a');
          a.href = url;
          a.download = kenvEntry.externalRef || `${kenvEntry.name}.kaspa`;
          a.click();

                                kenvEntry.password);
      }
  }

  function parseDirectoryEntries(columns, startIndex) {
      if (columns[startIndex] !== 'entries') return null;

      const fileCount = parseInt(columns[startIndex + 1]);
      const dirCount = parseInt(columns[startIndex + 2]);
      const entries = [];

      let index = startIndex + 3;
      const totalEntries = fileCount + dirCount;

      for (let i = 0; i < totalEntries; i++) {
          const type = columns[index] === 'f' ? 'file' : 'directory';
          const entry = {
              type: type,
              name: columns[index + 1],
              path: columns[index + 2] || '',  // 空文字列はルート
              size: parseInt(columns[index + 3]) || 0,
              metaTxId: columns[index + 4] || null,
              blockId: columns[index + 5] || null,
              password: columns[index + 6] || null
          };
          entries.push(entry);
          index += 7;  // 各エントリーは7カラム
      }

      return { entries, fileCount, dirCount };
  }
